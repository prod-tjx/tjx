# All endpoints live in one dict — a single --env flag at runtime picks prod or dev, same code path either way
IDMS_ENDPOINTS = {"prod": "https://idmsservice.corp.apple.com/apptoapp/token/generate", "dev": "https://idmsservice-uat.corp.apple.com/apptoapp/token/generate"}
ACM_ENDPOINTS  = {"prod": "acm-nonregulated.g.apple.com", "dev": "acm-nonregulated-uat2.g.apple.com"}

# Credential is injected by Jenkins as a secret binding at job runtime — it never touches the repo
app_password = os.environ.get("APP_PASSWORD")
if not app_password: raise RuntimeError("APP_PASSWORD env var required")  # fail loudly so silent auth failures never reach prod

# generate_a3_token posts APP_PASSWORD to IDMS, gets back a short-lived A3 token, then passes it to ACM
token = generate_a3_token(args.env, app_password)  # one call, right environment, no branching logic needed downstream
